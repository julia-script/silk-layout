cmake_minimum_required(VERSION 3.13)
project(yoga_harness CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release)
if(NOT YOGA_ROOT)
  message(FATAL_ERROR "set -DYOGA_ROOT to a facebook/yoga checkout")
endif()
include(${YOGA_ROOT}/cmake/project-defaults.cmake)
add_subdirectory(${YOGA_ROOT}/yoga ${CMAKE_CURRENT_BINARY_DIR}/yoga)
add_executable(yoga-harness harness.cpp)
target_link_libraries(yoga-harness yogacore)
target_include_directories(yoga-harness PRIVATE ${YOGA_ROOT}/lib)
